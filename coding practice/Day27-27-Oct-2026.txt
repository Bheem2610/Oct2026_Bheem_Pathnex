
variable "aws_region" {
  description = "AWS deployment region"
  type        = string
  default     = "ap-south-1"
}

variable "project_name" {
  type    = string
  default = "jenkins-cicd"
}

variable "admin_cidr" {
  description = "Your public IPv4 address in CIDR notation"
  type        = string

  validation {
    condition     = can(cidrhost(var.admin_cidr, 0))
    error_message = "Provide a valid CIDR block, such as 203.0.113.10/32."
  }
}

variable "key_name" {
  description = "Name of an existing AWS EC2 key pair"
  type        = string
}

variable "jenkins_instance_type" {
  type    = string
  default = "t3.medium"
}

variable "app_instance_type" {
  type    = string
  default = "t3.micro"
}

variable "app_public_key" {
  description = "Public SSH key installed for the deployment user"
  type        = string
}
